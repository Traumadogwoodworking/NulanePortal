(function () {
  function bySelector(selector) {
    return document.querySelector(selector);
  }

  function setStatus(value) {
    var status = bySelector("[data-auth0-bootstrap-status]");
    var state = bySelector("[data-auth0-bootstrap-state]");
    if (status) {
      status.textContent = value;
    }
    if (state) {
      state.textContent = value;
    }
    console.debug("[Auth0Bootstrap]", value);
  }

  function setError(err) {
    var error = bySelector("[data-auth0-bootstrap-error]");
    var message = err && err.message ? err.message : String(err || "unknown error");
    if (error) {
      error.textContent = message;
    }
    console.error("[Auth0Bootstrap]", err);
  }

  function getReturnTo() {
    try {
      return new URLSearchParams(window.location.search).get("returnTo") || "/home";
    } catch (error) {
      return "/home";
    }
  }

  function loadAuth0Script() {
    return new Promise(function (resolve, reject) {
      if (window.auth0 && window.auth0.createAuth0Client) {
        resolve();
        return;
      }

      setStatus("loading-auth0-script");

      var existing = document.querySelector("script[data-auth0-spa-js]");
      if (existing) {
        existing.addEventListener("load", function () {
          resolve();
        }, { once: true });
        existing.addEventListener("error", function () {
          reject(new Error("Auth0 script failed to load."));
        }, { once: true });
        return;
      }

      var script = document.createElement("script");
      script.src = "https://cdn.auth0.com/js/auth0-spa-js/2.6/auth0-spa-js.production.js";
      script.async = true;
      script.dataset.auth0SpaJs = "true";
      script.onload = function () {
        resolve();
      };
      script.onerror = function () {
        reject(new Error("Auth0 script failed to load."));
      };
      document.head.appendChild(script);
    });
  }

  async function startAuth0Login() {
    try {
      setStatus("script-started");

      var returnTo = getReturnTo();
      var redirectUri = window.location.origin;
      var domain = "nulanesystems.us.auth0.com";
      var clientId = "WkYT29HkNJo5rjDMPGTxAdb04QdKQsPc";
      var audience = "https://api.nulanesystems.com";

      if (!domain || !clientId || !audience) {
        throw new Error("Missing Auth0 domain/clientId/audience.");
      }

      var returnToNode = bySelector("[data-auth0-return-to]");
      if (returnToNode) {
        returnToNode.textContent = returnTo;
      }
      try {
        window.sessionStorage.setItem("portal_login_return_to", returnTo);
      } catch (storageError) {
        console.warn("[Auth0Bootstrap] unable to persist returnTo", storageError);
      }

      window.__portalAuth0LoginState = {
        returnTo: returnTo,
        redirectUri: redirectUri,
        domain: domain,
        clientId: clientId,
        audience: audience,
      };

      await loadAuth0Script();
      setStatus("script-loaded");

      if (!window.auth0 || !window.auth0.createAuth0Client) {
        throw new Error("Auth0 client is unavailable after loading the SDK.");
      }

      var client = await window.auth0.createAuth0Client({
        domain: domain,
        clientId: clientId,
        authorizationParams: {
          audience: audience,
          redirect_uri: redirectUri,
        },
        cacheLocation: "localstorage",
        useRefreshTokens: true,
      });

      setStatus("client-created");
      setStatus("redirecting");

      await client.loginWithRedirect({
        appState: { returnTo: returnTo },
        authorizationParams: {
          audience: audience,
          redirect_uri: redirectUri,
          prompt: "login",
        },
      });
    } catch (err) {
      setStatus("failed");
      setError(err);
      setStatus("failed: " + (err && err.message ? err.message : String(err || "unknown error")));
    }
  }

  window.__portalStartAuth0Login = startAuth0Login;

  var retryButton = bySelector("[data-auth0-bootstrap-retry]");
  if (retryButton) {
    retryButton.addEventListener("click", function () {
      void window.__portalStartAuth0Login();
    });
  }

  void window.__portalStartAuth0Login();
